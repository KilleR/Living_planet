namespace = lp_empire

# Calculate devouring growth rate
country_event = {
	id = lp_empire.001
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				count_situation = {
					limit = {
						is_situation_type = lp_situation_growth
					}
					count = 0
				}
			}
			lp_start_growth_situation = yes
		}

		every_owned_megastructure = {
			limit = {
				is_megastructure_type = planet_devourer
			}
			save_event_target_as = event_devourer_megastructure

			if = {
				limit = {
					NOT = {
						exists = planet
					}
				}
				remove_megastructure = event_target:event_devourer_megastructure
			}
		}
	}
}

# Check for Volcanic Terraforming eligibility
event = {
	id = lp_empire.010
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		any_country = {
			has_origin = origin_living_planet
			capital_scope.planet? = {
				is_planet_class = pc_volcanic
			}
		}
	}

	immediate = {
		every_country = {
			limit = {
				has_origin = origin_living_planet
				capital_scope.planet? = {
					is_planet_class = pc_volcanic
				}
				has_technology = tech_volcanic_terraforming
				NOT = {
					has_country_flag = lp_volcanic_terraforming
				}
			}

			country_event = {
				id = lp_empire.011
			}
		}
	}
}

# Grant Volcanic Terraforming
country_event = {
	id = lp_empire.011
	is_triggered_only = yes

	picture = GFX_evt_inf_cataclysmic_eruption
	title = lp_empire.011.name
	desc = lp_empire.011.desc

	show_sound = event_plasma_volcanoes
	
	option = {
		name = lp_empire.011.a
	}

	immediate = {
		set_country_flag = lp_volcanic_terraforming
	}

	after = {
		custom_tooltip = lp_empire.011.a.desc
		set_country_flag = allow_terraforming_into_volcanic
	}
}
