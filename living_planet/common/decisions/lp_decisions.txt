decision_lp_devour_planetary_matter = {
	icon = decision_planet_beam
	owned_planets_only = no
	
	potential = {
		is_colonizable = yes
		NOT = { has_modifier = living_planet_devoured }
		exists = space_owner
		space_owner = {
			is_same_value = from
			has_origin = origin_living_planet
		}
	}
	
	effect = {
		add_modifier = { modifier = living_planet_devoured }
		
		solar_system = {
			spawn_megastructure = {
				type = planet_devourer
				planet = ROOT
				owner = space_owner
			}
		}
	}
}

decision_lp_volcanise = {
	icon = decision_exploding_planet
	owned_planets_only = yes

	potential = {
		is_capital = yes
		is_planet_class = pc_volcanic
		hidden_trigger = {
			NOT = {
				has_deposit = d_active_volcano
				owner = {
					has_technology = tech_volcano
				}
			}
		}
		space_owner = {
			has_origin = origin_living_planet
			has_technology = tech_terrestrial_sculpting
		}
	}

	allow = {
	}

	effect = {
		owner = {
			add_tech_progress = {
				tech = tech_volcano
				progress = 0
			}
		}
		add_blocker = {
			type = d_active_volcano
		}
	}
}
